archetypes:
  - name: "Spacecraft and Artificers"
    colors: "WU"
    strategy: "Build a board of creatures with enter-the-battlefield abilities, then use them to station Spacecraft. Protect your vessels while they come online."
    cards: ["Constable Odo", "Interstellar Effigy"]
  - name: "Dilemma Control"
    colors: "UB"
    strategy: "Use modal spells to choose between proactive and reactive plays, control the opponent, and reward yourself for facing dilemmas."
    cards: ["Consider the Prime Directive", "Resistance Is Futile"]
  - name: "Klingons and Dies"
    colors: "BR"
    strategy: "Attack with hasty Klingon Warrior tokens, then sacrifice creatures for additional value. Some cards reward your creatures for dying."
    cards: ["A Good Day to Die", "Vengeance Mission", "Sarcophagus, Ship of the Dead", "Starfleet Security"]
  - name: "Away Team Landfall"
    colors: "RG"
    strategy: "Explore with Starfleet away teams and trigger landfall abilities. Some cards can double those triggers."
    cards: ["Captain Michael Burnham", "Jake Sisko, War Correspondent", "Dal R'El, Augment Orphan", "Erica Ortegas, Plucky Pilot"]
  - name: "Doctors and Counters"
    colors: "GW"
    strategy: "Support creatures with counters. Phlox cares about counters of any kind, including charge counters on Spacecraft."
    cards: ["Pure Energy", "Support Mission", "Phlox, Unorthodox Doctor"]
  - name: "Borg Artifact Creatures"
    colors: "WB"
    strategy: "Establish a board of artifact creatures, grow their power, and sacrifice artifacts when they have outlived their usefulness."
    cards: ["Vexing Drone", "Assault Drone", "Locutus, the One Who Speaks"]
  - name: "Scientists and Noncreatures"
    colors: "UR"
    strategy: "Use low-cost noncreature spells to trigger scientist payoffs. Spacecraft artifacts also count as noncreature spells."
    cards: ["Gray and Adira, Forever Joined", "Test of Priorities", "Test of Identity"]
  - name: "Archaeology Graveyard"
    colors: "BG"
    strategy: "Fill your graveyard with creatures and artifacts, then reuse them. Repeatable surveil helps stock the graveyard and find what you need."
    cards: ["Vash, Freelance Archaeologist", "Pattern Buffer Preservation"]
  - name: "Command 8+ Total Power"
    colors: "RW"
    strategy: "Build a wide board of Officers and reach eight or more total power. Clear blockers to let your team through."
    cards: ["Academy Instructor", "La'An Noonien-Singh, Security", "Worf, Honor-Bound Warrior"]
  - name: "Strange New Worlds"
    colors: "GU"
    strategy: "Explore differently named lands to support the strange new worlds ability word. Off-color Planet and Anomaly lands can still contribute."
    cards: ["The Traveler", "Suus Mahna Practitioner"]
